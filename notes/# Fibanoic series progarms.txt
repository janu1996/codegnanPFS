# perfect number

n = int(input("Enter a number: "))

sum = 0

for i in range(1, n):
    if n % i == 0:
        sum = sum + i

if sum == n:
    print("It is a perfect number")
else:
    print("It is not a perfect number")

 # palindrome for numbers

 n = int(input("Enter a number: "))

original = n
reverse = 0

while n > 0:
    digit = n % 10
    reverse = reverse * 10 + digit
    n = n // 10

if original == reverse:
    print("It is a palindrome number")
else:
    print("It is not a palindrome number")

# palindrome for string 

s = input("Enter a string: ")

reverse = ""

for i in range(len(s) - 1, -1, -1):
    reverse = reverse + s[i]

if s == reverse:
    print("It is a palindrome")
else:
    print("It is not a palindrome")

# using slicing

s = input("Enter a string: ")

reverse = s[::-1]

if s == reverse:
    print("It is a palindrome")
else:
    print("It is not a palindrome")

# Rescusive for (1 to n)

def print_numbers(n):
    if n == 0:
        return

    print_numbers(n - 1)
    print(n)

n = int(input("Enter n: "))
print_numbers(n)


Dry Run

When print_numbers(5) is called:

Step 1

print_numbers(5)
→ n is not 0
→ print_numbers(4)

Step 2

print_numbers(4)
→ n is not 0
→ print_numbers(3)

Step 3

print_numbers(3)
→ n is not 0
→ print_numbers(2)

Step 4

print_numbers(2)
→ n is not 0
→ print_numbers(1)

Step 5

print_numbers(1)
→ n is not 0
→ print_numbers(0)

Step 6

print_numbers(0)
→ n == 0
→ return

Now the function starts returning back.

Returning back
print_numbers(1)
→ print(1)  → 1


print_numbers(2)
→ print(2)  → 2


print_numbers(3)
→ print(3)  → 3


print_numbers(4)
→ print(4)  → 4


print_numbers(5)
→ print(5)  → 5
Final Output
1
2
3
4
5

types of recursion

1. direct
2. indirect --> not returning to the same function name.
example of indirect:
def even(n):
    if n==0:
        return True
    return odd(n-1)
def odd(n):
    if n==0:
        return False
    return even(n - 1)
print(even(5))
print(odd(4))

# fibanoic series using recursion:


def fib(n):
    if n<=1:
        return n
    return fib(n-1)+fib(n-2)
    
n=int(input("Enter a number:"))

    print(fib(i),end= " ")
