def linear_search(arr,target):
    for num in arr:
        if num == target:
            return True
    return False
arr =[10,5,18,25,30,55]
target = 25
result = linear_search(arr,target)
print(result)

2.find max element in arr 7 4 12 8 3.

code:
def maximum(arr):
    largest = arr[0]
    for num  in arr:
        if num>largest:
            largest = num
    return largest
arr =[7,4,12,8,3]
result = maximum(arr)
print(result)
