def linear_search(arr, target):
    for i in range(len(arr)):
        if arr[i] == target:
            return i

arr = [25, 44, 17, 12, 58, 41, 36, 98]
target = 12

result = linear_search(arr, target)
print("Target found at index:", result)

binary search 
algorithm:
step-1 : initilalize left and right
left = 0 
right=n-1

step-2: find middle
middle = left+right/2

step-3:if target ==mid: return target
step-4: if target>arr[mid] move -->right--->left =mid+1
step-5: if target<arr[mid] move-->left-->right=mid-1
